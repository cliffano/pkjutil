    /**
 * @module pkjutil
 */
"use strict";
import _ from "lodash";
import async from "async";
import "colors";
import fs from "fs";
import p from "path";
import semver from "semver";
import util from "util";

const HTTP_TIMEOUT = 30000;

const DEPENDENCY_TYPES = [
  "dependencies",
  "devDependencies",
  "peerDependencies",
  "optDependencies",
];

/**
 * Utility for reading and modifying a package.json file.
 */
class PkjUtil {
  /**
   * Constructor for initialising PkjUtil.
   *
   * @param {String} [file=package.json] - path to package.json file
   */
  constructor(file) {
    this.file = file || "package.json";
  }

  /**
   * List dependencies.
   *
   * @param {String} [type] - dependencies type (dependencies,
   *   devDependencies, peerDependencies, optDependencies), all types when
   *   not provided
   * @param {Function} cb - standard cb(err, result) callback
   */
  listDependencies(type, cb) {
    this._readPackage(function (err, pkg) {
      if (err) {
        cb(err);
      } else {
        const types = type ? [type] : DEPENDENCY_TYPES;
        let modules = [];
        _.filter(types, function (type) {
          return pkg[type];
        }).forEach(function (type) {
          modules = modules.concat(_.keys(pkg[type]));
        });
        modules = _.uniq(modules);
        cb(null, modules);
      }
    });
  }

  /**
   * Sort dependencies in alphabetical order.
   *
   * @param {String} [type] - dependencies type (dependencies,
   *   devDependencies, peerDependencies, optDependencies), all types when
   *   not provided
   * @param {Function} cb - standard cb(err, result) callback
   */
  sortDependencies(type, cb) {
    function process(pkg, cb) {
      const types = type ? [type] : DEPENDENCY_TYPES;
      _.filter(types, function (type) {
        return pkg[type];
      }).forEach(function (type) {
        const sorted = [];
        _.keys(pkg[type])
          .sort()
          .forEach(function (module) {
            sorted.push(util.format('"%s": "%s"', module, pkg[type][module]));
          });
        pkg[type] = JSON.parse(util.format("{%s}", sorted.join(",")));
      });
      cb(null, pkg);
    }

    this._updatePackage(process, cb);
  }

  /**
   * Traverse dependencies and group each package with its dependencies
   * reference.
   *
   * @param {Function} cb - standard cb(err, result) callback
   */
  traverseDependencies(cb) {
    if (fs.existsSync("package.json") || fs.existsSync("node_modules")) {
      this._readInstalled(process.cwd(), function (err, pkgs) {
        let result;
        if (!err) {
          result = {};
          pkgs.forEach(function (pkg) {
            // optional dependencies are also dependencies, as per npm
            const types = {
              dependencies: _.assign(
                {},
                pkg.dependencies,
                pkg.optionalDependencies,
              ),
              devDependencies: pkg.devDependencies,
              peerDependencies: pkg.peerDependencies,
              optDependencies: pkg.optDependencies,
            };
            const name = util.format("%s@%s", pkg.name, pkg.version);
            result[name] = {};
            _.filter(DEPENDENCY_TYPES, function (type) {
              return !_.isEmpty(types[type]);
            }).forEach(function (type) {
              result[name][type] = types[type];
            });
          });
        }
        cb(err, result);
      });
    } else {
      cb(
        new Error(
          "Either package.json file or node_modules directory must exist",
        ),
      );
    }
  }

  /**
   * Upgrade version based on type.
   *
   * @param {String} [type=patch] - version type (major, minor, patch)
   * @param {Function} cb - standard cb(err, result) callback
   */
  upgradeVersion(type, cb) {
    function process(pkg, cb) {
      const version = semver.parse(pkg.version);
      if (type === "major") {
        version.major += 1;
        version.minor = 0;
        version.patch = 0;
      } else if (type === "minor") {
        version.minor += 1;
        version.patch = 0;
      } else {
        version.patch += 1;
      }
      pkg.version = version.format();
      cb(null, pkg);
    }

    this._updatePackage(process, cb);
  }

  /**
   * Upgrade the version of each dependency to the latest published
   * in the registry.
   * Version range will be kept, e.g. ~0.0.1 is upgraded to ~0.0.18 .
   * If version does not contain numeric major.minor.patch, then
   * version is upgraded to latest, e.g. 0.0.x is upgraded to 0.0.18 .
   *
   * @param {Object} [opts] - optional settings
   * @param {String} [opts.registry=https://registry.npmjs.org/] - npm
   *   registry URL
   * @param {Function} cb - standard cb(err, result) callback
   */
  upgradeDependencies(opts, cb) {
    const LIMIT = 5;
    const self = this;

    opts = opts || {};
    opts.registry = opts.registry || "https://registry.npmjs.org/";
    if (!opts.registry.match(/\/$/)) {
      opts.registry += "/";
    }

    function process(pkg, cb) {
      function iterator(dependency, cb) {
        const url = opts.registry + dependency + "/latest";
        self._httpGet(url, function (err, body) {
          if (err) {
            cb(err);
            return;
          }
          let latestVersion;
          try {
            latestVersion = JSON.parse(body).version;
          } catch (parseErr) {
            cb(
              new Error(
                util.format(
                  "Unable to parse registry response for %s: %s",
                  dependency,
                  parseErr.message,
                ),
              ),
            );
            return;
          }

          function update(type) {
            const dependencies = pkg[type];
            const currVersionMatch = dependencies[dependency].match(
              /[0-9]+\.[0-9]+\.[0-9]+/,
            );
            const currVersion = currVersionMatch
              ? currVersionMatch.toString()
              : null;
            const range = currVersionMatch
              ? dependencies[dependency].replace(
                  /[0-9]+\.[0-9]+\.[0-9]+.*$/,
                  "",
                )
              : null;

            if (currVersion != latestVersion) {
              dependencies[dependency] = (range || "") + latestVersion;
              console.log(
                "%s - %s - upgraded to %s",
                dependency.green,
                type,
                dependencies[dependency],
              );
            } else {
              console.log(
                "%s - %s - already latest %s",
                dependency.grey,
                type,
                dependencies[dependency],
              );
            }
          }

          _.filter(DEPENDENCY_TYPES, function (type) {
            return pkg[type] && pkg[type][dependency];
          }).forEach(function (type) {
            update(type);
          });
          cb();
        });
      }

      let dependencies = [];
      DEPENDENCY_TYPES.forEach(function (type) {
        dependencies = _.union(dependencies, _.keys(pkg[type]));
      });

      async.eachLimit(dependencies, LIMIT, iterator, function (err) {
        cb(err, pkg);
      });
    }

    this._updatePackage(process, cb);
  }

  /**
   * Set the node engine version.
   *
   * @param {String} version - node engine version, e.g. >= 4.0
   * @param {Function} cb - standard cb(err, result) callback
   */
  setNodeEngine(version, cb) {
    function process(pkg, cb) {
      pkg.engines = pkg.engines || {};
      pkg.engines.node = version;
      cb(null, pkg);
    }

    this._updatePackage(process, cb);
  }

  /**
   * Read the package.json of a package directory and of every package
   * installed under its node_modules directory, recursively, including
   * scoped packages and nested node_modules directories.
   * Each package is read once, identified by its real path, so symlinked
   * packages (e.g. via npm link) are followed but not repeated.
   * Directories without package.json are skipped.
   *
   * @private
   * @param {String} dir - root package directory
   * @param {Function} cb - standard cb(err, result) callback, result is an
   *   array of parsed package.json objects, root package first
   */
  _readInstalled(dir, cb) {
    const seen = {};
    const pkgs = [];
    const MISSING = ["ENOENT", "ENOTDIR"];

    function visible(names) {
      return names.filter(function (name) {
        return !name.startsWith(".");
      });
    }

    function readNodeModules(nodeModulesDir, cb) {
      fs.readdir(nodeModulesDir, function (err, names) {
        if (err) {
          cb(MISSING.includes(err.code) ? null : err);
          return;
        }
        async.eachSeries(
          visible(names),
          function (name, next) {
            const entry = p.join(nodeModulesDir, name);
            if (!name.startsWith("@")) {
              readPackageDir(entry, next);
              return;
            }
            fs.readdir(entry, function (err, scopedNames) {
              if (err) {
                next(err);
                return;
              }
              async.eachSeries(
                visible(scopedNames),
                function (scopedName, nextScoped) {
                  readPackageDir(p.join(entry, scopedName), nextScoped);
                },
                next,
              );
            });
          },
          cb,
        );
      });
    }

    function readPackageDir(pkgDir, cb) {
      fs.realpath(pkgDir, function (err, realDir) {
        if (err) {
          cb(err);
          return;
        }
        if (seen[realDir]) {
          cb();
          return;
        }
        seen[realDir] = true;
        const file = p.join(realDir, "package.json");
        fs.readFile(file, "utf8", function (err, data) {
          if (err && !MISSING.includes(err.code)) {
            cb(err);
            return;
          }
          if (!err) {
            try {
              pkgs.push(JSON.parse(data));
            } catch (parseErr) {
              cb(
                new Error(
                  util.format("Unable to parse %s: %s", file, parseErr.message),
                ),
              );
              return;
            }
          }
          readNodeModules(p.join(realDir, "node_modules"), cb);
        });
      });
    }

    readPackageDir(dir, function (err) {
      if (err) {
        cb(err);
      } else {
        cb(null, pkgs);
      }
    });
  }

  /**
   * Send an HTTP GET request using Node.js built-in fetch.
   * Credentials in the URL (http://user:pass@host) are sent as a Basic
   * Authorization header, since fetch does not accept them in the URL.
   * Proxy environment variables are honoured when Node.js is run with
   * NODE_USE_ENV_PROXY=1.
   *
   * @private
   * @param {String} url - URL to request
   * @param {Function} cb - standard cb(err, result) callback, result is the
   *   response body when the status code is 200
   */
  _httpGet(url, cb) {
    const parsedUrl = new URL(url);
    const headers = {};
    if (parsedUrl.username || parsedUrl.password) {
      const credentials = util.format(
        "%s:%s",
        decodeURIComponent(parsedUrl.username),
        decodeURIComponent(parsedUrl.password),
      );
      headers.Authorization =
        "Basic " + Buffer.from(credentials).toString("base64");
      parsedUrl.username = "";
      parsedUrl.password = "";
    }

    fetch(parsedUrl.toString(), {
      headers: headers,
      signal: AbortSignal.timeout(HTTP_TIMEOUT),
    })
      .then(function (response) {
        return response.text().then(function (body) {
          if (response.status !== 200) {
            throw new Error(
              util.format(
                "Unexpected status code: %d\nResponse body:\n%s",
                response.status,
                body,
              ),
            );
          }
          return body;
        });
      })
      .then(
        function (body) {
          // call back outside the promise chain so that errors thrown by the
          // callback are not caught and passed to the callback a second time
          process.nextTick(cb, null, body);
        },
        function (err) {
          // fetch wraps network errors in a generic "fetch failed" TypeError,
          // the cause carries the system error e.g. getaddrinfo ENOTFOUND
          process.nextTick(cb, err.cause instanceof Error ? err.cause : err);
        },
      );
  }

  /**
   * Read and parse package.json file.
   *
   * @private
   * @param {Function} cb - standard cb(err, result) callback, result is the
   *   parsed package.json object
   */
  _readPackage(cb) {
    const self = this;
    fs.readFile(this.file, function (err, data) {
      if (err) {
        cb(err);
      } else {
        let pkg;
        try {
          pkg = JSON.parse(data);
        } catch (parseErr) {
          cb(
            new Error(
              util.format(
                "Unable to parse %s: %s",
                self.file,
                parseErr.message,
              ),
            ),
          );
          return;
        }
        cb(null, pkg);
      }
    });
  }

  /**
   * Read package.json file, apply process function, then overwrite the file.
   *
   * @private
   * @param {Function} process - process function to be applied to parsed
   *   package.json
   * @param {Function} cb - standard cb(err, result) callback
   */
  _updatePackage(process, cb) {
    const self = this;
    this._readPackage(function (err, pkg) {
      if (err) {
        cb(err);
      } else {
        process(pkg, function (err, result) {
          if (err) {
            cb(err, result);
          } else {
            fs.writeFile(
              self.file,
              JSON.stringify(result, null, 2) + "\n",
              function (err) {
                cb(err, result);
              },
            );
          }
        });
      }
    });
  }
}

export { PkjUtil as default };
